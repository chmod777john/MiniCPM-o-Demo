# O5 最简可见序列文法草案

## 终结符

```text
UNIT_START            = <unit>
UNIT_END              = </unit>

NON_INFER
SPOKEN

THINK_START           = <think>
THINK_END             = </think>

TOOL_CALL_START       = <tool-call>
TOOL_CALL_END         = </tool-call>
TOOL_ABORT            = <tool-abort>

TOOL_RESPONSE_START   = <tool-response>
TOOL_RESPONSE_END     = </tool-response>

BUDGET_REACHED        = <budget-reached>

THINK_TOKEN
TOOL_CALL_TOKEN
TOOL_RESPONSE_JSON_ID_RESULT
```

## 非终结符

```text
SESSION
UNIT
UNIT_PREFIX
UNIT_CONTENT
INFER_PROJECTION
INFER_NORMALIZATION
INFER_STREAM
INFER_TOKEN
IGNORE
TOP_ITEM
THINK_BLOCK
THINK_BODY
TOOL_CALL_BLOCK
TOOL_CALL_BODY
TOOL_RESPONSE_BLOCK
```

## 产生式

```text
SESSION :=
  UNIT+

UNIT :=
  UNIT_START UNIT_PREFIX UNIT_CONTENT UNIT_END

UNIT_PREFIX :=
  NON_INFER SPOKEN

UNIT_CONTENT :=
  INFER_TOKEN+

INFER_PROJECTION(SESSION) :=
  concat(UNIT_CONTENT for UNIT in SESSION)

INFER_NORMALIZATION :=
  erase_ignore(INFER_PROJECTION(SESSION))

INFER_NORMALIZATION ∈ INFER_STREAM

INFER_STREAM :=
  TOP_ITEM+

INFER_TOKEN :=
    THINK_START
  | THINK_END
  | TOOL_CALL_START
  | TOOL_CALL_END
  | TOOL_ABORT
  | TOOL_RESPONSE_START
  | TOOL_RESPONSE_END
  | BUDGET_REACHED
  | THINK_TOKEN
  | TOOL_CALL_TOKEN
  | TOOL_RESPONSE_JSON_ID_RESULT

IGNORE :=
  BUDGET_REACHED

TOP_ITEM :=
    THINK_BLOCK
  | TOOL_CALL_BLOCK
  | TOOL_RESPONSE_BLOCK

THINK_BLOCK :=
  THINK_START THINK_BODY THINK_END

THINK_BODY :=
  THINK_TOKEN*

TOOL_CALL_BLOCK :=
    TOOL_CALL_START TOOL_CALL_BODY TOOL_CALL_END
  | TOOL_CALL_START TOOL_CALL_BODY TOOL_ABORT

TOOL_CALL_BODY :=
  TOOL_CALL_TOKEN*

TOOL_RESPONSE_BLOCK :=
  TOOL_RESPONSE_START TOOL_RESPONSE_JSON_ID_RESULT TOOL_RESPONSE_END
```
